/* Pie mínimo del crédito tradicional desde el motor (el mismo que usa /simular). Si el motor no responde, queda el texto de respaldo (20%). El valor máximo solo sirve para esconder ejemplos que lo superen; no se muestra en la página. */
(function () {
  var root = document.getElementById("sv-root");
  if (!root || root.getAttribute("data-sv-ok")) return;
  root.setAttribute("data-sv-ok", "1");
  var MOTOR = "https://motor4life-944155992.development.catalystserverless.com/server/motor_4life/";
  var PIE = 0.20, VMAX = 30000, UF = 0;
  function nro(x) { return typeof x === "number" && x > 0; }
  function f0(n) { return Math.round(n).toLocaleString("es-CL"); }
  function set(k, v) { [].forEach.call(root.querySelectorAll('[data-sv="' + k + '"]'), function (e) { e.textContent = v; }); }
  function out(k) { return root.querySelector('[data-sv-out="' + k + '"]'); }
  function mm(uf) { var m = uf * UF / 1e6; return "unos $" + m.toLocaleString("es-CL", { maximumFractionDigits: m < 10 ? 1 : 0 }) + " millones"; }

  /* Calculadora del pie: valor elegido x pie mínimo */
  function pintar() {
    var r = root.querySelector('input[name="sv-valor"]:checked'); if (!r) return;
    var v = +r.value, pie = v * PIE, cred = v - pie;
    out("pie").textContent = f0(pie);
    out("cred").textContent = f0(cred);
    var bar = root.querySelector(".sv-bar-pie"); if (bar) bar.style.setProperty("--p", PIE);
    [["pie_clp", pie], ["cred_clp", cred]].forEach(function (x) { var e = out(x[0]); if (e && UF) { e.textContent = mm(x[1]); e.hidden = false; } });
  }
  [].forEach.call(root.querySelectorAll('input[name="sv-valor"]'), function (i) { i.addEventListener("change", pintar); });
  /* Valores por sobre el máximo del motor se esconden */
  function topar() { [].forEach.call(root.querySelectorAll('input[name="sv-valor"]'), function (i) { if (+i.value > VMAX) i.parentNode.hidden = true; }); }

  fetch(MOTOR + "limites?producto=tradicional").then(function (r) { return r.json(); }).then(function (j) {
    ((j && j.limites && j.limites.campos) || []).forEach(function (c) {
      if (c.id === "valor_uf" && nro(c.max)) VMAX = c.max;
      if (c.id === "pie_uf" && c.min && nro(c.min.pct) && c.min.pct < 1) PIE = c.min.pct;
    });
    set("pie", f0(PIE * 100));
    topar(); pintar();
  }).catch(function () {});

  /* Equivalencia en pesos con la UF del día desde el motor (GET /uf: CMF, con respaldo). Es la misma UF que usa /simular. Si el motor no responde o entrega su valor de respaldo, los pesos no se muestran. */
  fetch(MOTOR + "uf").then(function (r) { return r.json(); }).then(function (j) {
    if (j && nro(j.valor) && j.fuente !== "respaldo") { UF = j.valor; pintar(); }
  }).catch(function () {});
  pintar();
})();
</script>
