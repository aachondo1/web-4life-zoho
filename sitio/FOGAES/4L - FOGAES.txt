/* Cifras de FOGAES y del crédito tradicional desde el motor (mismos límites que usa /simular). Si el motor no responde, queda el texto de respaldo. */
(function () {
  var root = document.getElementById("fg-root");
  if (!root || root.getAttribute("data-fg-ok")) return;
  root.setAttribute("data-fg-ok", "1");
  var MOTOR = "https://motor4life-944155992.development.catalystserverless.com/server/motor_4life/";
  var EJEMPLO_UF = 3000;
  function nro(x) { return typeof x === "number" && x > 0; }
  function f0(n) { return Math.round(n).toLocaleString("es-CL"); }
  function lim(id) {
    return fetch(MOTOR + "limites?producto=" + id).then(function (r) { return r.json(); }).then(function (j) {
      var L = {};
      ((j && j.limites && j.limites.campos) || []).forEach(function (c) {
        if (c.id === "valor_uf" && nro(c.max)) L.vmax = c.max;
        if (c.id === "pie_uf" && c.min && nro(c.min.pct)) L.pie = c.min.pct;
      });
      return L;
    });
  }
  function set(k, v) { [].forEach.call(root.querySelectorAll('[data-fg="' + k + '"]'), function (e) { e.textContent = v; }); }
  Promise.all([lim("fogaes"), lim("tradicional")]).then(function (r) {
    var F = r[0], T = r[1];
    if (F.vmax) set("valor_max", f0(F.vmax));
    if (F.pie && F.pie < 1) { set("pie_fogaes", f0(F.pie * 100)); set("fin_fogaes", f0(100 - F.pie * 100)); set("ej_fogaes", f0(EJEMPLO_UF * F.pie)); }
    if (T.pie && T.pie < 1) { set("pie_trad", f0(T.pie * 100)); set("fin_trad", f0(100 - T.pie * 100)); set("ej_trad", f0(EJEMPLO_UF * T.pie)); }
    var pf = F.pie || 0.10, pt = T.pie || 0.20;
    pintar(pf, pt);
  }).catch(function () {});
  var DIF = EJEMPLO_UF * 0.10;
  function pintar(pf, pt) {
    DIF = EJEMPLO_UF * (pt - pf);
    set("ej_dif", f0(DIF));
    var h = root.querySelector("[data-fg-h]"), bars = root.querySelector("[data-fg-bars]");
    if (h) h.style.setProperty("--h", pf / pt);
    if (bars) bars.setAttribute("aria-label", "Ejemplo: para una vivienda de UF " + f0(EJEMPLO_UF) + ", el pie es UF " + f0(EJEMPLO_UF * pf) + " con FOGAES y UF " + f0(EJEMPLO_UF * pt) + " con crédito tradicional: UF " + f0(DIF) + " menos");
    pesos();
  }
  /* Equivalencia en pesos con la UF del día desde el motor (GET /uf: CMF, con respaldo), la misma que usa /simular. Si no responde o es el valor de respaldo, no se muestra. */
  var UF = 0;
  function pesos() {
    var e = root.querySelector("[data-fg-clp]");
    if (!e || !UF || DIF <= 0) return;
    var m = DIF * UF / 1e6;
    e.textContent = ", unos $" + m.toLocaleString("es-CL", { maximumFractionDigits: m < 10 ? 1 : 0 }) + " millones";
    e.hidden = false;
  }
  fetch(MOTOR + "uf").then(function (r) { return r.json(); }).then(function (j) {
    if (j && nro(j.valor) && j.fuente !== "respaldo") { UF = j.valor; pesos(); }
  }).catch(function () {});
})();
</script>
