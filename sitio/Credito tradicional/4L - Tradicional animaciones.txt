<script>
(function () {
  if (window.matchMedia && matchMedia("(prefers-reduced-motion: reduce)").matches) return;
  document.documentElement.classList.add("ct-anim");
  /* Así funciona: se decide después de cargar (y de saltar a un #ancla); si la sección ya está a la vista, queda quieta */
  function pasos() {
    var root = document.getElementById("ct-root"), road = root && root.querySelector(".ct-road"), stops = road && road.querySelector(".v-stops");
    if (!stops || !("IntersectionObserver" in window) || road.getAttribute("data-anim")) return;
    road.setAttribute("data-anim", "1");
    var r = stops.getBoundingClientRect();
    if (r.top < window.innerHeight && r.bottom > 0) return;
    root.classList.add("ct-road-anim");
    var io = new IntersectionObserver(function (es) {
      es.forEach(function (e) {
        var h = e.rootBounds ? e.rootBounds.height : window.innerHeight;
        if (e.isIntersecting && (e.intersectionRatio >= 0.5 || e.intersectionRect.height >= 0.5 * h)) { road.classList.add("is-drawn"); io.disconnect(); }
      });
    }, { threshold: [0, 0.25, 0.5, 1] });
    io.observe(stops);
  }
  function later() { requestAnimationFrame(function () { requestAnimationFrame(pasos); }); }
  if (document.readyState === "complete") later(); else window.addEventListener("load", later);
})();
</script>
