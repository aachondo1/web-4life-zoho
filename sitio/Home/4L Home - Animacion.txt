<script>
(function () {
  var road = document.querySelector(".f4h .h-road");
  if (!road || road.getAttribute("data-anim")) return;
  road.setAttribute("data-anim", "1");
  var root = road.closest(".f4h"), stops = road.querySelector(".v-stops");
  /* Sin soporte o con "reducir movimiento": queda el diseño final, sin animar */
  if (!stops || !("IntersectionObserver" in window)) return;
  if (window.matchMedia && matchMedia("(prefers-reduced-motion: reduce)").matches) return;
  function decide() {
    /* Se decide después de cargar (y de saltar a un #ancla): si la sección ya está a la vista, no se anima */
    var r = stops.getBoundingClientRect();
    if (r.top < window.innerHeight && r.bottom > 0) return;
    root.classList.add("f4h-anim");
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (e) {
        var h = e.rootBounds ? e.rootBounds.height : window.innerHeight;
        if (e.isIntersecting && (e.intersectionRatio >= 0.5 || e.intersectionRect.height >= 0.5 * h)) { road.classList.add("is-drawn"); io.disconnect(); }
      });
    }, { threshold: [0, 0.25, 0.5, 1] });
    io.observe(stops);
  }
  function later() { requestAnimationFrame(function () { requestAnimationFrame(decide); }); }
  if (document.readyState === "complete") later(); else window.addEventListener("load", later);
})();
</script>
