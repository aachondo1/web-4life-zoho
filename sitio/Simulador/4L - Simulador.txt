  function enviar(accion, extra) {
    var p = {
      accion: accion, hp: $("s4-web").value,
      nombre: persona.nombre, apellido_paterno: persona.apellido, apellido_materno: persona.apellido2 || "",
      rut: persona.rut, email: persona.email, edad: persona.edad
    };
    /* Consentimiento: se envía solo en el primer evento después de marcar las casillas (no en cada paso).
       Si la persona vuelve a enviar el formulario de datos, se marca pendiente de nuevo.
       El CRM igual deja nota solo si el consentimiento es nuevo o cambia la versión del texto legal. */
    if (persona.consentPendiente) {
      p.acepta = persona.acepta; p.acepta_marketing = !!persona.marketing; p.version_tyc = VERSION_TYC;
      persona.consentPendiente = false;
    }
    if (persona.telefono) p.telefono = persona.telefono;
    Object.keys(extra || {}).forEach(function (k) { p[k] = extra[k]; });
    /* UTM y página de origen: primero los guardados en la sesión (f4_utm); los de la URL actual mandan */
    try {
      var utm = JSON.parse(sessionStorage.getItem("f4_utm") || "{}");
      Object.keys(utm).forEach(function (k) { p[k] = utm[k]; });
      var q = new URLSearchParams(location.search);
      ["utm_source","utm_medium","utm_campaign","utm_term","utm_content","gclid","fbclid"].forEach(function (k) {
        if (q.get(k)) p[k] = q.get(k);
      });
      p.origen = q.get("origen") || location.pathname.replace(/^\//, "");
    } catch (e) {}
    /* Cookie _fbp del Pixel de Meta (solo existe si la persona aceptó cookies de marketing) */
    try { var fbpM = document.cookie.match(/(?:^|; )_fbp=([^;]+)/); if (fbpM) p.fbp = fbpM[1]; } catch (e) {}
    if (!ENDPOINT) { try { console.info("[modo prueba] enviaría a Zoho:", p); } catch (e) {} return; }
    try {
      fetch(ENDPOINT, { method: "POST", mode: "no-cors", keepalive: true, headers: { "Content-Type": "text/plain;charset=UTF-8" }, body: JSON.stringify(p) }).catch(function () {});
    } catch (e) {}
  }

  /* Respuesta del motor. Un 400 trae "detalle" (lista de textos) o "disponibles" (producto desconocido); 401, 500 y red no. */
  function leerRespuesta(res) {
    if (res.status === 400) return res.json().catch(function () { return {}; }).then(function (j) { throw errMotor(j); });
    if (!res.ok) throw new Error("HTTP " + res.status);
    return res.json();
  }
  /* Cada texto de detalle: "campo: mensaje" (límites) o "campo mensaje" (tipo u obligatorio).
     El campo es lo que va antes del primer ":" o espacio. */
  function errMotor(j) {
    var e = new Error("HTTP 400"), d = j && j.detalle;
    e.v400 = true; e.disp = j && j.disponibles;
    e.errs = (d && d.map ? d : d ? [d] : []).map(function (t) {
      t = String(t); var m = t.match(/^(\w+)(:?)\s*([\s\S]*)$/), x = m && m[2] ? m[3] : t.replace(/_uf\b/, "").replace(/_/g, " ");
      return { campo: m ? m[1] : "", msg: x.charAt(0).toUpperCase() + x.slice(1) };
    });
    return e;
  }

  function pedirComp() {
    var p = sel, g = compGen, pieT = pieTrad();
    if (PRODUCTO !== "fogaes" || !pieT || base.pie < pieT || !(p > 0) || comp.hasOwnProperty(p)) return;
    comp[p] = null;
    var bt = { producto: "tradicional", valor_uf: base.valor, pie_uf: base.pie, tipo_propiedad: base.tipo, edad: persona.edad, plazo_anios: p };
    if (radio("ingresos")) bt.n_deudores = radio("ingresos") === "codeudor" ? 2 : 1;
    calcularMotor(bt)
      .then(function (j) {
        if (g !== compGen) return;
        var t = (j.plazos || []).filter(function (x) { return x.plazo_anios === p; })[0];   /* el plazo pedido */
        t = t ? adaptar(t, j) : null;
        if (!t) throw new Error("sin plazo " + p);
        comp[p] = t;
        if (sel === p) pintarComp();
      })
      .catch(function () { if (g === compGen) delete comp[p]; });   /* se reintenta al volver a mostrar ese plazo */
  }
